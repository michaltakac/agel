; The dock now appears and returns after reboot.
; Repaint it through a native actor, with no rebuild or host renderer.
(def dock-behavior (fn (self state message) (begin (dock-paint message) message)))
(def dock (spawn dock-behavior 3423048))
(send dock 4609905)
